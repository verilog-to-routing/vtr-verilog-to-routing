###############################################################################
# Configuration file for running the MCNC benchmarks through the AP flow.
#
# This sweeps the candidate gain attenuation functions supported by APPack
# (--appack_gain_attenuation_fn). Unlike the other MCNC-based strong_ap tests,
# the circuits here are not fixed to a device and their IOs are not
# constrained -- only the channel width is fixed, so this also exercises
# APPack's cluster growth on an auto-sized device.
###############################################################################

# Path to directory of circuits to use
circuits_dir=benchmarks/blif/wiremap6

# Path to directory of architectures to use
archs_dir=arch/timing

# Add architectures to list to sweep
arch_list_add=k6_frac_N10_40nm.xml

# Add circuits to list to sweep
circuit_list_add=apex4.pre-vpr.blif
circuit_list_add=des.pre-vpr.blif
circuit_list_add=seq.pre-vpr.blif

# Constrain the circuits to their channel widths
#       1.3 * minW
circuit_constraint_list_add=(apex4.pre-vpr.blif,    route_chan_width=78)
circuit_constraint_list_add=(seq.pre-vpr.blif,      route_chan_width=78)
circuit_constraint_list_add=(des.pre-vpr.blif,      route_chan_width=44)

# Parse info and how to parse
parse_file=vpr_fixed_chan_width.txt

# How to parse QoR info
qor_parse_file=qor_ap_fixed_chan_width.txt

# Pass requirements
pass_requirements_file=pass_requirements_ap_fixed_chan_width.txt

# Pass the script params while writing the vpr constraints.
script_params_common=-starting_stage vpr -track_memory_usage --analytical_place --ap_full_legalizer appack --route --analysis

# Sweep the gain attenuation functions supported by APPack.
script_params_list_add=--appack_gain_attenuation_fn none
script_params_list_add=--appack_gain_attenuation_fn quad_sqrt_knee
script_params_list_add=--appack_gain_attenuation_fn gaussian
