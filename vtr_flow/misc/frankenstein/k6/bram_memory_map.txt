# bram_memory_map.txt
# fallback static for k6. normally regenerated from the arch xml by
# vtr_arch_rules. keep the wide widths  the bit-slice techmap needs more
# than 1-bit modes or every memory collapses to 32kx1 tiles.
# sp 512x64 1024x32 2048x16 4096x8 8192x4 16384x2 32768x1
# dp 1024x32 2048x16 2048x8 8192x4 16384x2 32768x1  (2048x8 is half capacity)

ram block $__VTR_SP_RAM__ {
  abits 15;
  widths 1 2 4 8 16 32 64 per_port;
  cost 30;
  init none;

  port srsw "A" {
    clock posedge;
    # undefined so both read-first and write-first $mem cells match. safe
    # when a memory never reads and writes the same address in one cycle.
    rdwr undefined;
    wrtrans all old;
  }
}

ram block $__VTR_DP_RAM__ {
  abits 15;
  widths 1 2 4 8 16 32 per_port;
  cost 100;
  init none;

  # shared clock domain so both ports sit on one physical clock
  port srsw "A" "B" {
    clock posedge "C";
    rdwr undefined;
    wrtrans all old;
    optional_rw;
  }
}
